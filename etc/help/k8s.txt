# Load Image into Minikube
minikube image load smart-lottery-service:1.0-SNAPSHOT

# List Minikube Images
minikube image ls

# Create Deployment
kubectl create deployment smart-lottery-service-deployment --image=smart-lottery-service:1.0-SNAPSHOT
kubectl apply -f deployment.yaml

# Delete Deployment
kubectl delete deployment smart-lottery-service-deployment

# Get Deployments
kubectl get deployments

# Get Pods
kubectl get pods

# Describe Pod
kubectl describe pod smart-lottery-service-deployment-6fcf7bbc84-j46pv

# Pod's Logs
kubectl logs smart-lottery-service-deployment-6fcf7bbc84-j46pv

# Get Interactive Terminal for Pod
kubectl exec -it smart-lottery-service-deployment-6fcf7bbc84-j46pv -- bin/bash
kubectl exec -it smart-lottery-service-deployment-6fcf7bbc84-j46pv -- sh

# Create/Expose NodePort Service for Deployment
kubectl expose deployment smart-lottery-service-deployment --type=NodePort --port=8081 --target-port=8081

# Get Services
kubectl get services

# Get External URL for Deployment
minikube service smart-lottery-service-deployment  --url

# Delete Service
kubectl delete service smart-lottery-service-deployment
kubectl delete svc smart-lottery-service-deployment

# Port-Forwarding
kubectl port-forward deployment/smart-lottery-service-deployment 8081:8081
kubectl port-forward pod/smart-lottery-service-deployment-6fcf7bbc84-j46pv 8081:8081
