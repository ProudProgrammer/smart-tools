# Describe Node
kubectl describe node

# Create Deployment
kubectl create deployment smart-lottery-service-deployment --image=smart-lottery-service:1.0-SNAPSHOT
kubectl apply -f smart-lottery-service-deployment.yaml

# Delete Deployment
kubectl delete deployment smart-lottery-service-deployment

# Get Deployments
kubectl get deployments
kubectl get deployment smart-lottery-service-deployment -o yaml

# Get Pods
kubectl get pods
kubectl get pods -o wide
kubectl get pods --all-namespaces
kubectl get pods -n smart-platform
kubectl get pods -n smart-platform -o wide

# Describe Pod
kubectl describe pod smart-lottery-service-deployment-6fcf7bbc84-j46pv

# Pod's Logs
kubectl logs smart-lottery-service-deployment-6fcf7bbc84-j46pv

# Get Interactive Terminal for Pod
kubectl exec -it smart-lottery-service-deployment-6fcf7bbc84-j46pv -- bin/bash
kubectl exec -it smart-lottery-service-deployment-6fcf7bbc84-j46pv -- sh

# Create/Expose NodePort Service for Deployment to reach Pod from outside the Cluster
kubectl expose deployment smart-lottery-service-deployment --type=NodePort --port=8081 --target-port=8081

# Create NodePort Service to reach Pod from outside the Cluster
kubectl apply -f smart-lottery-service-service.yaml

# Get Services
kubectl get services

# Describe Service
kubectl describe service smart-lottery-service-service

# Get External URL for Deployment
minikube service smart-lottery-service-deployment  --url
minikube service smart-lottery-service-service
minikube service smart-edge-service-service
minikube service smart-lottery-service-service --url

# Delete Service
kubectl delete service smart-lottery-service-deployment
kubectl delete svc smart-lottery-service-deployment

# Port-Forwarding
kubectl port-forward deployment/smart-lottery-service-deployment 8081:8081
kubectl port-forward pod/smart-lottery-service-deployment-6fcf7bbc84-j46pv 8081:8081
kubectl port-forward service/smart-edge-service-service 8080:8080 -n smart-platform
kubectl port-forward service/smart-lottery-service-service 8081:8081 -n smart-platform
kubectl port-forward service/smart-ui-service 8001:8001 -n smart-platform

# Get All
kubectl get all
kubectl get all --all-namespaces
kubectl get all -o wide

# Delete All
kubectl delete all --all
kubectl delete configmap smart-platform-configmap

# Create Namespace
kubectl create namespace smart-platform

# Set Default Namespace
kubectl config set-context --current --namespace=smart-platform
